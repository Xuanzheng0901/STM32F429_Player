cmake_minimum_required(VERSION 3.22)
project(music_ui_tests C)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PORT "${REPO}/User/components/display/lvgl_port")
file(READ "${PORT}/lv_conf.h" CONFIG)
# 64 位主机对象含更宽的指针；固件仍使用 24 KiB LVGL 堆。
string(REPLACE "LV_MEM_SIZE (24 * 1024U)" "LV_MEM_SIZE (32 * 1024U)" CONFIG "${CONFIG}")
string(REPLACE "LV_USE_OS   LV_OS_FREERTOS" "LV_USE_OS   LV_OS_NONE" CONFIG "${CONFIG}")
string(REPLACE "LV_ASSERT_HANDLER while(1);" "LV_ASSERT_HANDLER abort();" CONFIG "${CONFIG}")
string(REPLACE "LV_ASSERT_HANDLER_INCLUDE <stdint.h>" "LV_ASSERT_HANDLER_INCLUDE <stdlib.h>" CONFIG "${CONFIG}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h" "${CONFIG}")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/lvgl" lvgl)
include("${PORT}/lvgl_sources.cmake")
get_target_property(SOURCES lvgl SOURCES)
list(FILTER SOURCES EXCLUDE REGEX "/src/osal/lv_freertos[.]c$")
list(APPEND SOURCES "${REPO}/lvgl/src/osal/lv_os_none.c")
set_property(TARGET lvgl PROPERTY SOURCES "${SOURCES}")
add_executable(music_ui test_ui.c "${REPO}/User/components/UI/music_player_assets.c"
    "${PORT}/fusion_pixel_12_cjk.c")
target_include_directories(music_ui PRIVATE mocks "${REPO}/User/components/player"
    "${REPO}/User/components/UI" "${REPO}/User/components/fatfs" "${REPO}/User/components/audio"
    "${REPO}/User/components/display" "${REPO}/User/components/display/stm_ssd1685" "${PORT}")
target_compile_definitions(music_ui PRIVATE portMAX_DELAY=0xffffffffU)
target_compile_options(music_ui PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(music_ui PRIVATE lvgl m)
add_test(NAME music_ui COMMAND music_ui)

add_executable(encoder test_encoder.c "${PORT}/fusion_pixel_12_cjk.c")
target_include_directories(encoder PRIVATE mocks "${PORT}")
target_compile_options(encoder PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(encoder PRIVATE lvgl m)
add_test(NAME encoder COMMAND encoder)
